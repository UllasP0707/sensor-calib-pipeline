cmake_minimum_required(VERSION 3.20)
project(sensor_calib_pipeline VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_ASAN "Enable AddressSanitizer instrumentation" OFF)
option(BUILD_CALIBRATION "Build the OpenCV camera calibration engine" ON)
option(BUILD_SIMULATION "Build batched highway simulation" ON)
option(ENABLE_CUDA "Build the NVIDIA CUDA simulation backend" OFF)
include(CTest)
include(FetchContent)
find_package(Threads REQUIRED)
function(calib_options target)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-O3>)
        if(ENABLE_ASAN)
            target_compile_options(${target} PRIVATE -fsanitize=address -fno-omit-frame-pointer -g)
            target_link_options(${target} PRIVATE -fsanitize=address)
        endif()
    elseif(ENABLE_ASAN)
        message(FATAL_ERROR "ENABLE_ASAN requires GCC or Clang")
    endif()
endfunction()
if(BUILD_TESTING)
    find_package(GTest QUIET)
    if(NOT GTest_FOUND)
        FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git
            GIT_TAG v1.14.0 GIT_SHALLOW TRUE)
        FetchContent_MakeAvailable(googletest)
    endif()
    include(GoogleTest)
endif()
if(BUILD_CALIBRATION)
    find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs calib3d highgui)
    find_package(nlohmann_json 3.11 QUIET)
    if(NOT nlohmann_json_FOUND)
        FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git
            GIT_TAG v3.11.3 GIT_SHALLOW TRUE)
        FetchContent_MakeAvailable(json)
    endif()
    add_library(calib_core src/calib_engine.cpp)
    target_include_directories(calib_core PUBLIC include)
    target_include_directories(calib_core SYSTEM PUBLIC ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(calib_core PUBLIC ${OpenCV_LIBS} nlohmann_json::nlohmann_json Threads::Threads)
    add_executable(calib_engine src/main.cpp)
    target_link_libraries(calib_engine PRIVATE calib_core)
    calib_options(calib_core)
    calib_options(calib_engine)
    if(BUILD_TESTING)
        add_executable(test_calib_engine tests/test_calib_engine.cpp)
        target_link_libraries(test_calib_engine PRIVATE calib_core GTest::gtest_main)
        calib_options(test_calib_engine)
        gtest_discover_tests(test_calib_engine DISCOVERY_TIMEOUT 60)
    endif()
    install(TARGETS calib_engine RUNTIME DESTINATION bin)
endif()
if(BUILD_SIMULATION)
    add_library(truck_sim SHARED sim/src/sim_api.cpp sim/src/sim_cpu.cpp)
    target_include_directories(truck_sim PUBLIC sim/include)
    target_link_libraries(truck_sim PRIVATE Threads::Threads)
    calib_options(truck_sim)
    if(ENABLE_CUDA)
        if(ENABLE_ASAN)
            message(FATAL_ERROR "Use separate CPU ASan and CUDA compute-sanitizer builds")
        endif()
        enable_language(CUDA)
        find_package(CUDAToolkit REQUIRED)
        target_sources(truck_sim PRIVATE sim/src/sim_cuda.cu)
        target_compile_definitions(truck_sim PRIVATE SIM_ENABLE_CUDA)
        target_link_libraries(truck_sim PRIVATE CUDA::cudart)
        set_target_properties(truck_sim PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
        target_compile_options(truck_sim PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;-lineinfo>)
    endif()
    if(BUILD_TESTING)
        add_executable(test_sim sim/tests/test_sim.cpp)
        target_link_libraries(test_sim PRIVATE truck_sim GTest::gtest_main)
        calib_options(test_sim)
        gtest_discover_tests(test_sim DISCOVERY_TIMEOUT 60)
    endif()
    install(TARGETS truck_sim LIBRARY DESTINATION lib)
endif()
